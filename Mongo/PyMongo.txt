from pymongo import MongoClient

# Establish connection to MongoDB
mongo_client = MongoClient("mongodb://localhost:27017/")

# Get database and collection names from user
database_name = input("Enter database name: ")
my_database = mongo_client[database_name]

collection_name = input("Enter collection name: ")
my_collection = my_database[collection_name]

print(f"\n✓ Collection '{collection_name}' is ready to use!\n")

# Main program loop
while True:
    print("="*50)
    print("        PRODUCT MANAGEMENT SYSTEM")
    print("="*50)
    print("[1] Add New Product")
    print("[2] View All Products")
    print("[3] Find Product by ID")
    print("[4] Remove Product by ID")
    print("[5] Exit Program")
    print("="*50)
    
    try:
        user_choice = int(input("Select an option (1-5): "))
    try:
        user_choice = int(input("Select an option (1-5): "))
    except ValueError:
        print("\n⚠ Invalid input! Please enter a number between 1-5.\n")
        continue

    if user_choice == 1:
        # Insert new product
        print("\n--- Add New Product ---")
        product_id = int(input("Product ID: "))
        product_name = input("Product Name: ")
        product_price = float(input("Product Price (₹): "))
        
        new_product = {
            "pid": product_id,
            "pname": product_name,
            "pprice": product_price
        }
        my_collection.insert_one(new_product)
        print(f"\n✓ Product '{product_name}' added successfully!\n")

    elif user_choice == 2:
        # Display all products
        print("\n" + "="*50)
        print("           ALL PRODUCTS")
        print("="*50)
        
        total_products = my_collection.count_documents({})
        if total_products == 0:
            print("No products available in the database.")
        else:
            for product in my_collection.find():
                print(f"ID: {product['pid']} | Product: {product['pname']} | Price: ₹{product['pprice']}")
        print("="*50 + "\n")

    elif user_choice == 3:
        # Search product by ID
        search_id = int(input("\nEnter Product ID to search: "))
        found_product = my_collection.find_one({"pid": search_id})
        
        if found_product:
            print("\n✓ Product Details:")
            print(f"  ID: {found_product['pid']}")
            print(f"  Name: {found_product['pname']}")
            print(f"  Price: ₹{found_product['pprice']}\n")
        else:
            print(f"\n✗ No product found with ID: {search_id}\n")

    elif user_choice == 4:
        # Delete product by ID
        delete_id = int(input("\nEnter Product ID to delete: "))
        deletion_result = my_collection.delete_one({"pid": delete_id})
        
        if deletion_result.deleted_count > 0:
            print(f"\n✓ Product with ID {delete_id} deleted successfully!\n")
        else:
            print(f"\n✗ No product found with ID: {delete_id}\n")

    elif user_choice == 5:
        # Exit program
        print("\n" + "="*50)
        print("   Thank you for using Product Management System!")
        print("="*50)
        break

    else:
        print("\n⚠ Invalid option! Please select between 1-5.\n")
